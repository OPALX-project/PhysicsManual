67 std::shared_ptr<PC_t> makeContainer(ParticleType species = ParticleType::UNNAMED) {
68 ippl::Vector<int, 3> nr = 8;
69 ippl::Vector<double, 3> rmin = -4.0;
70 ippl::Vector<double, 3> rmax = 4.0;
71 ippl::Vector<double, 3> origin = rmin;
72 ippl::Vector<double, 3> hr = (rmax - rmin) / ippl::Vector<double, 3>(nr);
73 std::array<bool, 3> decomp = {true, true, true};
74
75 ippl::NDIndex<3> domain;
76 for (unsigned i = 0; i < 3; ++i) {
77 domain[i] = ippl::Index(nr[i]);

79
80 Mesh_t<3> mesh(domain, hr, origin);
81 FieldLayout_t<3> fl(MPI_COMM_WORLD, domain, decomp, true);
82 auto pc = std::make_shared<PC_t>(mesh, fl);
83 pc->Sp = static_cast<short>(species);
84 pc->setBunchStateHandler(std::make_shared<BunchStateHandler>());
85 return pc;

110 TEST_F(PionDecaySpectrumTest, PionDecayRestFrameMomentumIsFixed) {
111 constexpr std::size_t nPions = 100000;
112 constexpr std::size_t nBins = 50;
113
114 auto pions = makeContainer();
115 auto muons = makeContainer(ParticleType::MUON);
116 pions->setM(Physics::m_pi);
117 muons->setM(Physics::m_mu);
118 createParticles(pions, nPions, 0.0);
119
120 Options::seed = 20260516;
121 PionDecay decay(1.0e-12, 0, Physics::m_pi);
122 decay.setDaughterContainer(muons, Physics::m_mu);
123
124 const std::size_t marked = decay.apply(*pions, 1.0, 0, 0);
125 const std::size_t destroyed = pions->deleteInvalidParticles();
126 ASSERT_EQ(marked, nPions);
127 ASSERT_EQ(destroyed, nPions);
128 ASSERT_EQ(muons->getLocalNum(), nPions);
129
130 auto muP_host = muons->P.getHostMirror();
131 Kokkos::deep_copy(muP_host, muons->P.getView());
132
133 const double pFixed = (Physics::m_pi * Physics::m_pi - Physics::m_mu * Physics::m_mu)
134 / (2.0 * Physics::m_pi);
135
136 std::vector<double> cosThetaSamples;
137 cosThetaSamples.reserve(nPions);
138 double maxRelDev = 0.0;
139 for (std::size_t i = 0; i < nPions; ++i) {
140 const double bgx = muP_host(i)[0];
141 const double bgy = muP_host(i)[1];
142 const double bgz = muP_host(i)[2];
143 const double pmag = Physics::m_mu * std::sqrt(bgx * bgx + bgy * bgy + bgz * bgz);
144 maxRelDev = std::max(maxRelDev, std::abs(pmag - pFixed) / pFixed);
145 cosThetaSamples.push_back(Physics::m_mu * bgz / pmag);
146 }
147 std::cout << "[PionDecayRestFrame] |p|=" << pFixed << " max rel dev=" << maxRelDev << '\n';
148 EXPECT_LT(maxRelDev, 1.0e-12);
149
150 const auto hist = opalx::test::makeHistogram(cosThetaSamples, -1.0, 1.0, nBins);
151 const std::vector<double> uniform(nBins, 0.5);
152
153 const double l1 = opalx::test::l1Distance(hist, uniform);
154 const double mean = opalx::test::sampleMean(cosThetaSamples);
155 const double area = opalx::test::histogramArea(hist);
156 std::cout << "[PionDecayRestFrame] cos(theta) L1=" << l1 << " mean=" << mean
157 << " area=" << area << '\n';
158
159 EXPECT_LT(l1, 0.10);
160 EXPECT_NEAR(mean, 0.0, 0.02);
161 EXPECT_NEAR(area, 1.0, 0.01);
162
163 const std::string csvPath = "pion_rest_costheta.csv";
164 opalx::test::writeSpectrumCsv(csvPath, hist, uniform, "cos(theta) (rest frame)");
165 std::cout << "[PionDecayRestFrame] CSV written: " << csvPath << '\n';
166 }

168 TEST_F(PionDecaySpectrumTest, PionDecayBoostedLabEnergyIsFlatBox) {
169 constexpr std::size_t nPions = 100000;
170 constexpr std::size_t nBins = 50;
171 constexpr double parentBetaGam = 5.0; // pz of pion in beta*gamma units
172
173 auto pions = makeContainer();
174 auto muons = makeContainer(ParticleType::MUON);
175 pions->setM(Physics::m_pi);
176 muons->setM(Physics::m_mu);
177 createParticles(pions, nPions, parentBetaGam);
178
179 Options::seed = 20260517;
185 PionDecay decay(1.0e-12, 0, Physics::m_pi);
186 decay.setDaughterContainer(muons, Physics::m_mu);
187
188 const std::size_t marked = decay.apply(*pions, 1.0, 0, 0);
189 const std::size_t destroyed = pions->deleteInvalidParticles();
190 ASSERT_EQ(marked, nPions);
191 ASSERT_EQ(destroyed, nPions);
192 ASSERT_EQ(muons->getLocalNum(), nPions);
193
194 auto muP_host = muons->P.getHostMirror();
195 Kokkos::deep_copy(muP_host, muons->P.getView());
196
197 // Rest-frame two-body kinematics.
198 const double M = Physics::m_pi;
199 const double md = Physics::m_mu;
200 const double pStar = (M * M - md * md) / (2.0 * M);
201 const double eStar = (M * M + md * md) / (2.0 * M);
202
203 // Parent boost.
204 const double gamma = std::sqrt(1.0 + parentBetaGam * parentBetaGam);
205 const double betaGam = parentBetaGam; // = beta * gamma
206 const double eMinus = gamma * eStar - betaGam * pStar;
207 const double ePlus = gamma * eStar + betaGam * pStar;
208 const double uniform = 1.0 / (ePlus - eMinus);
209
210 std::vector<double> labEnergies;
211 labEnergies.reserve(nPions);
212 for (std::size_t i = 0; i < nPions; ++i) {
213 const double bg2 = muP_host(i)[0] * muP_host(i)[0] + muP_host(i)[1] * muP_host(i)[1]
214 + muP_host(i)[2] * muP_host(i)[2];
215 labEnergies.push_back(std::sqrt(1.0 + bg2) * Physics::m_mu);
216 }
217
218 const auto hist = opalx::test::makeHistogram(labEnergies, eMinus, ePlus, nBins);
219 const std::vector<double> analyticPdf(nBins, uniform);
220
221 const double l1 = opalx::test::l1Distance(hist, analyticPdf);
222 const double mean = opalx::test::sampleMean(labEnergies);
223 const double meanRef = 0.5 * (eMinus + ePlus);
224 const double area = opalx::test::histogramArea(hist);
225 std::cout << "[PionDecayBoosted] E in [" << eMinus << ", " << ePlus << "] GeV, L1=" << l1
226 << " mean=" << mean << " (analytic=" << meanRef << ") area=" << area << '\n';
227
228 EXPECT_LT(l1, 0.10);
229 EXPECT_NEAR(mean, meanRef, 0.005 * (ePlus - eMinus));
230 EXPECT_NEAR(area, 1.0, 0.01);
231
232 const std::string csvPath = "pion_boosted_energy_box.csv";
233 opalx::test::writeSpectrumCsv(csvPath, hist, analyticPdf, "E_mu^lab [GeV]");
234 std::cout << "[PionDecayBoosted] CSV written: " << csvPath << '\n';
235 }
