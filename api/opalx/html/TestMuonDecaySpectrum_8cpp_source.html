104 double michelNormalization(double a) { return 1.0 - 2.0 * a * a * a + a * a * a * a; }
105
108 double michelMean(double a) {
109 const double a4 = a * a * a * a;
110 const double a5 = a4 * a;
111 const double num = (1.5 - 0.8) - (1.5 * a4 - 0.8 * a5);
112 return num / michelNormalization(a);
113 }
